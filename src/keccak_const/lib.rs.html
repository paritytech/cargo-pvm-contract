1//! `const fn` implementation of the SHA-3 family of hash and extendable-output
2//! functions.
3//!
4//! This crate allows you to use the SHA-3 hash and extendable-output functions
5//! as constant expressions in Rust. For all other usages, the [sha3](https://crates.io/crates/sha3)
6//! crate includes more optimized implementations of these hash functions.
7//!
8//! # Examples
9//!
10//! ```rust
11//! # use keccak_const::Shake256;
12//! const PSEUDO_RANDOM_BYTES: [u8; 1000] = Shake256::new()
13//!         .update(b"The quick brown fox ")
14//!         .update(b"jumps over the lazy dog")
15//!         .finalize();
16//! ```
17//!
18//! ```rust
19//! # use keccak_const::Shake128;
20//! const ROUND_CONSTANTS: [u128; 8] = {
21//!     let shake = Shake128::new()
22//!         .update(b"The quick brown fox ")
23//!         .update(b"jumps over the lazy dog");
24//!
25//!     let mut reader = shake.finalize_xof();
26//!     let mut output = [0; 8];
27//!
28//!     let mut i = 0;
29//!     while i < 8 {
30//!         let buf: [u8; 16];
31//!         (reader, buf) = reader.read();
32//!         output[i] = u128::from_be_bytes(buf);
33//!         i += 1;
34//!     }
35//!
36//!     output
37//! };
38//!
39//! assert_eq!(
40//!     [
41//!         324498722242859095401832112442782838951,
42//!         100470442341479765851591908475476895342,
43//!         241049111671168257801898223573666863059,
44//!         139197826094415251816510671569090212218,
45//!         73371475849610774600276735485442220492,
46//!         321031806373587100556524628628207173306,
47//!         70553598458795679727810425741185559539,
48//!         297273966300911440566694043047331846682,
49//!     ],
50//!     ROUND_CONSTANTS,
51//! );
52//! ```

61const PADDING_SHA3: u8 = 0x06;
62const PADDING_KECCAK: u8 = 0x01;

64macro_rules! sha3 {
65    (
66        $(#[$doc:meta])* $name:ident,
67        $security:literal,
68        $padding:expr,
69    ) => {
70        $(#[$doc])*
71        #[derive(Clone)]
72        pub struct $name {
73            state: KeccakState,
74        }
75
76        impl $name {
77            /// Constructs a new hasher
78            pub const fn new() -> $name {
79                $name {
80                    state: KeccakState::new($security, $padding),
81                }
82            }
83
84            /// Absorbs additional input
85            ///
86            /// Can be called multiple times
87            pub const fn update(mut self, input: &[u8]) -> Self {
88                // usee `mut self` instead of `&mut self` because
89                // mutable references are unstable in constants.
90                self.state = self.state.update(input);
91                self
92            }
93
94            /// Pads and squeezes the state to the output
95            pub const fn finalize(&self) -> [u8; {$security / 8}] {
96                let reader = self.state.finalize();
97                let (_, output) = reader.read::<{$security / 8}>();
98                output
99            }
100        }
101    };
102}

316macro_rules! shake {
317    (
318        $(#[$doc:meta])* $name:ident,
319        $security:literal,
320    ) => {
321        $(#[$doc])*
322        pub struct $name {
323            state: KeccakState,
324        }
325
326        impl $name {
327            /// Constructs a new hasher
328            pub const fn new() -> $name {
329                $name {
330                    state: KeccakState::new($security, 0x1f),
331                }
332            }
333
334            /// Absorbs additional input
335            ///
336            /// Can be called multiple times.
337            pub const fn update(mut self, input: &[u8]) -> Self {
338                // use `mut self` instead of `&mut self` because
339                // mutable references are unstable in constants.
340                self.state = self.state.update(input);
341                self
342            }
343
344            /// Retrieves an extendable-output function (XOF) reader for current hasher instance
345            pub const fn finalize_xof(&self) -> XofReader {
346                self.state.finalize()
347            }
348
349            /// Finalizes the context and compute the output
350            pub const fn finalize<const N: usize>(&self) -> [u8; N] {
351                let reader = self.finalize_xof();
352                let (_, output) = reader.read::<N>();
353                output
354            }
355        }
356
357        impl Default for $name {
358            fn default() -> Self {
359                $name::new()
360            }
361        }
362    };
363}
