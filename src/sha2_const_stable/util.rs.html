1/// Copies `n` bytes from `src` (starting at `src_offset`) to `dest` (starting
2/// at `dest_offset`).
3///
4/// The source and destination must _not_ overlap. This function exists because
5/// subslices are not supported in `const fn`.
6#[inline(always)]
7pub(crate) const fn memcpy<const N: usize>(
8    mut dest: [u8; N],
9    dest_offset: usize,
10    src: &[u8],
11    src_offset: usize,
12    n: usize,
13) -> [u8; N] {
14    let mut i = 0;
15    while i < n {
16        dest[dest_offset + i] = src[src_offset + i];
17        i += 1;
18    }
19    dest
20}

22/// Sets `n` bytes in `dest` (starting at `dest_offset`) to `val`.
23///
24/// This function exists because subslices are not supported in `const fn`.
25#[inline(always)]
26pub(crate) const fn memset<const N: usize>(
27    mut dest: [u8; N],
28    offset: usize,
29    val: u8,
30    n: usize,
31) -> [u8; N] {
32    let mut i = 0;
33    while i < n {
34        dest[offset + i] = val;
35        i += 1;
36    }
37    dest
38}

40/// Loads an unsigned 32-bit big endian integer from `src` (starting at
41/// `offset`).
42///
43/// This function exists because subslices are not supported in `const fn`.
44#[inline(always)]
45pub(crate) const fn load_u32_be(src: &[u8], offset: usize) -> u32 {
46    u32::from_be_bytes([
47        src[offset],
48        src[offset + 1],
49        src[offset + 2],
50        src[offset + 3],
51    ])
52}

72/// Stores an unsigned 32-bit big endian integer into `dest` (starting at
73/// `offset`).
74///
75/// This function exists because subslices are not supported in `const fn`.
76#[inline(always)]
77pub(crate) const fn store_u32_be<const N: usize>(dest: [u8; N], offset: usize, n: u32) -> [u8; N] {
78    let bytes = u32::to_be_bytes(n);
79    memcpy(dest, offset, &bytes, 0, bytes.len())
80}
