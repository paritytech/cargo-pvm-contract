142/// Compute the keccak256 hash of a string
143fn keccak256(input: &str) -> [u8; 32] {
144    let mut hasher = Keccak::v256();
145    let mut output = [0u8; 32];
146    hasher.update(input.as_bytes());
147    hasher.finalize(&mut output);
148    output
149}
150
151/// Compute the 4-byte function selector from a function signature
152fn compute_selector(signature: &str) -> [u8; 4] {
153    let hash = keccak256(signature);
154    [hash[0], hash[1], hash[2], hash[3]]
155}

157/// Build a function signature from name and input types
158fn build_function_signature(name: &str, inputs: &[AbiInput]) -> String {
159    let types: Vec<&str> = inputs.iter().map(|i| i.type_name.as_str()).collect();
160    format!("{}({})", name, types.join(","))
161}

163/// Format a byte array as Rust hex literal
164fn format_bytes_as_hex(bytes: &[u8]) -> String {
165    bytes
166        .iter()
167        .map(|b| format!("0x{:02x}", b))
168        .collect::<Vec<_>>()
169        .join(", ")
170}

172/// Format a 32-byte array with line breaks for readability
173fn format_bytes32_multiline(bytes: &[u8; 32]) -> String {
174    bytes
175        .chunks(8)
176        .map(|chunk| {
177            chunk
178                .iter()
179                .map(|b| format!("0x{:02x}", b))
180                .collect::<Vec<_>>()
181                .join(", ")
182        })
183        .collect::<Vec<_>>()
184        .join(",\n    ")
185}

187/// Create a new blank contract project.
188pub fn init_blank_contract(contract_name: &str) -> Result<()> {
189    let contract_name = contract_name.to_case(Case::Kebab);
190    let target_dir = std::env::current_dir()?.join(&contract_name);
191    if target_dir.exists() {
192        anyhow::bail!("Directory already exists: {target_dir:?}");
193    }
194
195    fs::create_dir(&target_dir)
196        .with_context(|| format!("Failed to create directory: {target_dir:?}"))?;
197
198    let (target_json_path, target_json_name) = resolve_target_json()?;
199    let target_json_dest = target_dir.join(target_json_name);
200    fs::copy(&target_json_path, &target_json_dest).with_context(|| {
201        format!(
202            "Failed to copy target JSON from {} to {}",
203            target_json_path.display(),
204            target_json_dest.display()
205        )
206    })?;
207
208    let target_json_name = target_json_dest
209        .file_name()
210        .and_then(|name| name.to_str())
211        .ok_or_else(|| anyhow::anyhow!("Target JSON path is missing a file name"))?;
212
213    let cargo_config_dir = target_dir.join(".cargo");
214    fs::create_dir(&cargo_config_dir)?;
215    fs::write(
216        cargo_config_dir.join("config.toml"),
217        format!(
218            "[build]\n target = \"{}\"\n\n[unstable]\n build-std = [\"core\", \"alloc\"]\n\n[env]\n RUSTC_BOOTSTRAP = \"1\"\n",
219            target_json_name
220        ),
221    )?;
222
223    fs::write(target_dir.join(".gitignore"), "/target\n*.polkavm\n")?;
224    fs::write(
225        target_dir.join("rust-toolchain.toml"),
226        "[toolchain]\nchannel = \"nightly\"\n",
227    )?;
228    fs::create_dir(target_dir.join("src"))?;
229    let lib_rs_content = generate_blank_contract()?;
230    fs::write(
231        target_dir.join(format!("src/{}.rs", contract_name)),
232        lib_rs_content,
233    )?;
234
235    let build_rs_content = generate_build_rs()?;
236    fs::write(target_dir.join("build.rs"), build_rs_content)?;
237
238    let cargo_toml_content = generate_cargo_toml(&contract_name, &contract_name, false)?;
239    fs::write(target_dir.join("Cargo.toml"), cargo_toml_content)?;
240
241    println!("Successfully initialized blank contract project: {target_dir:?}");
242    println!("\nNext steps:");
243    println!("  cd {contract_name}");
244    println!("  cargo build");
245    Ok(())
246}

248/// Create a new contract project from a Solidity file.
249pub fn init_from_solidity_file(sol_file: &str, contract_name: &str, use_alloc: bool) -> Result<()> {
250    let sol_path = PathBuf::from(sol_file);
251    if !sol_path.exists() {
252        anyhow::bail!("Solidity file not found: {sol_file}");
253    }
254
255    let sol_abs_path = sol_path
256        .canonicalize()
257        .with_context(|| format!("Failed to get absolute path for {sol_file}"))?;
258
259    let sol_file_name = sol_path
260        .file_name()
261        .and_then(|s| s.to_str())
262        .ok_or_else(|| anyhow::anyhow!("Invalid file name"))?
263        .to_string();
264
265    let sol_content = fs::read(&sol_abs_path)
266        .with_context(|| format!("Failed to read Solidity file: {sol_abs_path:?}"))?;
267
268    init_from_example_files_inner(&sol_content, &sol_file_name, None, contract_name, use_alloc)
269}

287fn init_from_example_files_inner(
288    sol_contents: &[u8],
289    sol_file_name: &str,
290    rust_contents: Option<&[u8]>,
291    contract_name: &str,
292    use_alloc: bool,
293) -> Result<()> {
294    let contract_name = contract_name.to_case(Case::Kebab);
295    let sol_file_name = sol_file_name.to_string();
296
297    log::debug!("Extracting metadata from {sol_file_name}");
298    let (metadata, actual_contract_name) =
299        extract_solc_metadata_from_bytes(sol_contents, &sol_file_name)?;
300    let actual_contract_kebab = actual_contract_name.to_case(Case::Kebab);
301
302    // Create project directory
303    let target_dir = std::env::current_dir()?.join(&contract_name);
304    if target_dir.exists() {
305        anyhow::bail!("Directory already exists: {target_dir:?}");
306    }
307    fs::create_dir(&target_dir)
308        .with_context(|| format!("Failed to create directory: {target_dir:?}"))?;
309
310    let (target_json_path, target_json_name) = resolve_target_json()?;
311    let target_json_dest = target_dir.join(target_json_name);
312    fs::copy(&target_json_path, &target_json_dest).with_context(|| {
313        format!(
314            "Failed to copy target JSON from {} to {}",
315            target_json_path.display(),
316            target_json_dest.display()
317        )
318    })?;
319
320    let target_json_name = target_json_dest
321        .file_name()
322        .and_then(|name| name.to_str())
323        .ok_or_else(|| anyhow::anyhow!("Target JSON path is missing a file name"))?;
324
325    // Copy .sol file to project
326    let target_sol_path = target_dir.join(&sol_file_name);
327    fs::write(&target_sol_path, sol_contents)
328        .with_context(|| format!("Failed to write {sol_file_name} to {target_sol_path:?}"))?;
329
330    // Create .cargo directory and config
331    let cargo_config_dir = target_dir.join(".cargo");
332    fs::create_dir(&cargo_config_dir)?;
333    fs::write(
334        cargo_config_dir.join("config.toml"),
335        format!(
336            "[build]\n target = \"{}\"\n\n[unstable]\n build-std = [\"core\", \"alloc\"]\n\n[env]\n RUSTC_BOOTSTRAP = \"1\"\n",
337            target_json_name
338        ),
339    )?;
340
341    // Create .gitignore
342    fs::write(target_dir.join(".gitignore"), "/target\n*.polkavm\n")?;
343    fs::write(
344        target_dir.join("rust-toolchain.toml"),
345        "[toolchain]\nchannel = \"nightly\"\n",
346    )?;
347    // Generate src/{contract}.rs
348    fs::create_dir(target_dir.join("src"))?;
349
350    let lib_rs_content = if let Some(contents) = rust_contents {
351        String::from_utf8(contents.to_vec()).context("Example Rust file is not valid UTF-8")?
352    } else if use_alloc {
353        generate_rust_code_alloc(&sol_file_name, &metadata, &actual_contract_name)?
354    } else {
355        generate_rust_code_no_alloc(&metadata, &actual_contract_name)?
356    };
357    fs::write(
358        target_dir.join(format!("src/{}.rs", actual_contract_kebab)),
359        lib_rs_content,
360    )?;
361
362    let build_rs_content = generate_build_rs()?;
363    fs::write(target_dir.join("build.rs"), build_rs_content)?;
364
365    // Create Cargo.toml
366    let cargo_toml_content =
367        generate_cargo_toml(&contract_name, &actual_contract_kebab, use_alloc)?;
368    fs::write(target_dir.join("Cargo.toml"), cargo_toml_content)?;
369
370    println!("Successfully initialized contract project from {sol_file_name}: {target_dir:?}");
371    println!("\nNext steps:");
372    println!("  cd {contract_name}");
373    println!("  cargo build");
374    Ok(())
375}

377/// Internal helpers for template generation.
378fn extract_solc_metadata_from_bytes(
379    sol_contents: &[u8],
380    sol_file_name: &str,
381) -> Result<(ContractMetadata, String)> {
382    let sol_content =
383        String::from_utf8(sol_contents.to_vec()).context("Solidity file is not valid UTF-8")?;
384
385    let solc_input = serde_json::json!({
386        "language": "Solidity",
387        "sources": {
388            sol_file_name: {
389                "content": sol_content
390            }
391        },
392        "settings": {
393            "outputSelection": {
394                "*": {
395                    "*": ["metadata"]
396                }
397            }
398        }
399    });
400
401    let solc_input_str = serde_json::to_string(&solc_input)?;
402
403    let mut child = Command::new("solc")
404        .arg("--standard-json")
405        .stdin(std::process::Stdio::piped())
406        .stdout(std::process::Stdio::piped())
407        .stderr(std::process::Stdio::piped())
408        .spawn()
409        .context("Failed to spawn solc. Make sure solc is installed and in PATH.")?;
410
411    child
412        .stdin
413        .as_mut()
414        .ok_or_else(|| anyhow::anyhow!("Failed to open stdin"))?
415        .write_all(solc_input_str.as_bytes())?;
416
417    let output_result = child
418        .wait_with_output()
419        .context("Failed to wait for solc")?;
420
421    if !output_result.status.success() {
422        let stderr = String::from_utf8_lossy(&output_result.stderr);
423        anyhow::bail!("solc failed: {stderr}");
424    }
425
426    log::debug!(
427        "solc stdout: {}",
428        String::from_utf8_lossy(&output_result.stdout)
429    );
430
431    let solc_output: SolcOutput =
432        serde_json::from_slice(&output_result.stdout).with_context(|| {
433            format!(
434                "Failed to parse solc output. Output was: {}",
435                String::from_utf8_lossy(&output_result.stdout)
436            )
437        })?;
438
439    // Extract metadata from the first contract
440    let contracts_for_file = solc_output
441        .contracts
442        .get(sol_file_name)
443        .ok_or_else(|| anyhow::anyhow!("No contract found in solc output"))?;
444
445    let (contract_name, contract_info) = contracts_for_file
446        .iter()
447        .next()
448        .ok_or_else(|| anyhow::anyhow!("No contract found in solc output"))?;
449
450    let metadata: ContractMetadata = serde_json::from_str(&contract_info.metadata)
451        .context("Failed to parse contract metadata")?;
452
453    Ok((metadata, contract_name.clone()))
454}

468fn generate_rust_code_alloc(

470    metadata: &ContractMetadata,

474
475    let functions: Vec<AllocFunctionInfo> = metadata

478        .iter()
479        .filter_map(|item| match item {
480            AbiItem::Function { name, .. } => Some(AllocFunctionInfo {
481                name: name.clone(),
482                name_snake: name.to_case(Case::Snake),
483                call_type: format!("{contract_name_pascal}::{name}Call"),
484            }),
485            _ => None,
486        })
487        .collect();
488
489    let template = ContractAllocTemplate {
490        sol_file_name,
491        functions,
492    };
493
494    template.render().context("Failed to render alloc template")
495}

497fn generate_rust_code_no_alloc(metadata: &ContractMetadata, contract_name: &str) -> Result<String> {
498    let contract_name_upper = contract_name.to_uppercase();
499
500    // Collect function selectors
501    let mut selectors = Vec::new();
502    let mut functions = Vec::new();
503
504    for item in &metadata.output.abi {
505        if let AbiItem::Function { name, inputs, .. } = item {
506            let signature = build_function_signature(name, inputs);
507            let selector = compute_selector(&signature);
508            let const_name = format!("{}_SELECTOR", name.to_case(Case::UpperSnake));
509
510            selectors.push(SelectorConst {
511                const_name: const_name.clone(),
512                bytes_hex: format_bytes_as_hex(&selector),
513                signature: signature.clone(),
514            });
515
516            // Generate decode params
517            let mut params = Vec::new();
518
519            for (idx, input) in inputs.iter().enumerate() {
520                let param_name = if input.name.is_empty() {
521                    format!("param_{}", idx)
522                } else {
523                    input.name.to_case(Case::Snake)
524                };
525
526                let decode_line =
527                    format!("// TODO: decode {param_name} of type {}", input.type_name);
528
529                params.push(ParamDecode { decode_line });
530            }
531
532            functions.push(NoAllocFunctionInfo {
533                name: name.clone(),
534                selector_const: const_name,
535                min_call_data_len: 4 + inputs.len() * 32,
536                params,
537            });
538        }
539    }
540
541    // Collect events
542    let events: Vec<EventConst> = metadata
543        .output
544        .abi
545        .iter()
546        .filter_map(|item| {
547            if let AbiItem::Event { name, inputs } = item {
548                let signature = build_function_signature(name, inputs);
549                let hash = keccak256(&signature);
550                Some(EventConst {
551                    const_name: format!("{}_EVENT_SIGNATURE", name.to_case(Case::UpperSnake)),
552                    bytes_hex: format_bytes32_multiline(&hash),
553                    signature,
554                })
555            } else {
556                None
557            }
558        })
559        .collect();
560
561    // Collect errors
562    let errors: Vec<ErrorConst> = metadata
563        .output
564        .abi
565        .iter()
566        .filter_map(|item| {
567            if let AbiItem::Error { name, inputs } = item {
568                let signature = build_function_signature(name, inputs);
569                let selector = compute_selector(&signature);
570                Some(ErrorConst {
571                    const_name: format!("{}_ERROR", name.to_case(Case::UpperSnake)),
572                    bytes_hex: format_bytes_as_hex(&selector),
573                    signature,
574                })
575            } else {
576                None
577            }
578        })
579        .collect();
580
581    let template = ContractNoAllocTemplate {
582        contract_name_upper: &contract_name_upper,
583        selectors,
584        events,
585        errors,
586        functions,
587    };
588
589    template
590        .render()
591        .context("Failed to render no-alloc template")
592}

594fn resolve_target_json() -> Result<(PathBuf, String)> {
595    let mut args = polkavm_linker::TargetJsonArgs::default();
596    args.is_64_bit = true;
597    let target_json = polkavm_linker::target_json_path(args)
598        .map_err(|e| anyhow::anyhow!("Failed to get target JSON: {e}"))?;
599
600    let target_name = target_json
601        .file_name()
602        .and_then(|name| name.to_str())
603        .ok_or_else(|| anyhow::anyhow!("Target JSON path is missing a file name"))?
604        .to_string();
605
606    Ok((target_json, target_name))
607}

609fn generate_cargo_toml(contract_name: &str, bin_source: &str, use_alloc: bool) -> Result<String> {
610    let builder_path = std::env::var("CARGO_PVM_CONTRACT_BUILDER_PATH")
611        .ok()
612        .filter(|value| !value.trim().is_empty());
613
614    if let Some(ref path) = builder_path {
615        let path = std::path::Path::new(path);
616        if !path.exists() {
617            anyhow::bail!("Builder path does not exist: {}", path.display());
618        }
619    }
620
621    let template = CargoTomlTemplate {
622        contract_name,
623        bin_source,
624        use_alloc,
625        builder_version: BUILDER_VERSION,
626        builder_path,
627    };
628    template
629        .render()
630        .context("Failed to render Cargo.toml template")
