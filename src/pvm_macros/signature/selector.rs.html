3pub fn compute_selector(canonical_signature: &str) -> [u8; 4] {
4    let mut hasher = Keccak::v256();
5    hasher.update(canonical_signature.as_bytes());
6    let mut output = [0u8; 32];
7    hasher.finalize(&mut output);
8    [output[0], output[1], output[2], output[3]]
9}
