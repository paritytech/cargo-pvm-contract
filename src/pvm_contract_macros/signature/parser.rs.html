3#[derive(Debug, Clone)]
4pub struct FunctionSignature {
5    pub name: String,
6    pub inputs: Vec<SolType>,
7    pub outputs: Vec<SolType>,
8}
9
10impl FunctionSignature {
11    pub fn parse(sig: &str) -> Result<Self, String> {
12        let sig = sig.trim();
13
14        let (main_part, outputs) = if let Some(idx) = sig.find(" returns ") {
15            let (main, ret) = sig.split_at(idx);
16            let ret = ret.trim_start_matches(" returns ").trim();
17            (main, Self::parse_type_list(ret)?)
18        } else if let Some(idx) = sig.find("returns(") {
19            let (main, ret) = sig.split_at(idx);
20            let ret = ret.trim_start_matches("returns").trim();
21            (main.trim(), Self::parse_type_list(ret)?)
22        } else {
23            (sig, vec![])
24        };
25
26        let open_paren = main_part
27            .find('(')
28            .ok_or_else(|| format!("Missing '(' in signature: {}", sig))?;
29        let close_paren = main_part
30            .rfind(')')
31            .ok_or_else(|| format!("Missing ')' in signature: {}", sig))?;
32
33        let name = main_part[..open_paren].trim().to_string();
34        if name.is_empty() {
35            return Err("Empty function name".to_string());
36        }
37
38        let params_str = &main_part[open_paren + 1..close_paren];
39        let inputs = Self::parse_params(params_str)?;
40
41        Ok(FunctionSignature {
42            name,
43            inputs,
44            outputs,
45        })
46    }
47
48    fn parse_type_list(s: &str) -> Result<Vec<SolType>, String> {
49        let s = s.trim();
50        if !s.starts_with('(') || !s.ends_with(')') {
51            return Err(format!("Type list must be wrapped in parentheses: {}", s));
52        }
53        Self::parse_params(&s[1..s.len() - 1])
54    }
55
56    fn parse_params(params_str: &str) -> Result<Vec<SolType>, String> {
57        let params_str = params_str.trim();
58        if params_str.is_empty() {
59            return Ok(vec![]);
60        }
61
62        let mut types = vec![];
63        let mut depth = 0;
64        let mut current = String::new();
65
66        for ch in params_str.chars() {
67            match ch {
68                '(' => {
69                    depth += 1;
70                    current.push(ch);
71                }
72                ')' => {
73                    depth -= 1;
74                    current.push(ch);
75                }
76                ',' if depth == 0 => {
77                    let ty = Self::parse_single_type(current.trim())?;
78                    types.push(ty);
79                    current.clear();
80                }
81                _ => current.push(ch),
82            }
83        }
84
85        if !current.trim().is_empty() {
86            let ty = Self::parse_single_type(current.trim())?;
87            types.push(ty);
88        }
89
90        Ok(types)
91    }
92
93    fn parse_single_type(s: &str) -> Result<SolType, String> {
94        let s = s.trim();
95
96        if s.starts_with('(') {
97            let close = Self::find_matching_paren(s, 0)?;
98            let tuple_inner = &s[1..close];
99            let tuple_types = Self::parse_params(tuple_inner)?;
100
101            let rest = &s[close + 1..];
102            return Self::apply_array_suffixes(SolType::Tuple(tuple_types), rest);
103        }
104
105        if let Some(bracket_start) = s.find('[') {
106            let base_type_str = &s[..bracket_start];
107            let base_type = Self::parse_base_type(base_type_str)?;
108            return Self::apply_array_suffixes(base_type, &s[bracket_start..]);
109        }
110
111        Self::parse_base_type(s)
112    }
113
114    fn parse_base_type(s: &str) -> Result<SolType, String> {
115        let s = s.trim();
116
117        match s {
118            "address" => Ok(SolType::Address),
119            "bool" => Ok(SolType::Bool),
120            "string" => Ok(SolType::String),
121            "bytes" => Ok(SolType::DynBytes),
122            _ if s.starts_with("uint") => {
123                let bits: usize = s[4..].parse().unwrap_or(256);
124                if bits == 0 || bits > 256 || bits % 8 != 0 {
125                    return Err(format!("Invalid uint size: {}", bits));
126                }
127                Ok(SolType::Uint(bits))
128            }
129            _ if s.starts_with("int") => {
130                let bits: usize = s[3..].parse().unwrap_or(256);
131                if bits == 0 || bits > 256 || bits % 8 != 0 {
132                    return Err(format!("Invalid int size: {}", bits));
133                }
134                Ok(SolType::Int(bits))
135            }
136            _ if s.starts_with("bytes") => {
137                let size: usize = s[5..]
138                    .parse()
139                    .map_err(|_| format!("Invalid bytes size: {}", s))?;
140                if size == 0 || size > 32 {
141                    return Err(format!("Invalid bytes size: {}", size));
142                }
143                Ok(SolType::Bytes(size))
144            }
145            _ => Err(format!("Unknown type: {}", s)),
146        }
147    }
148
149    fn apply_array_suffixes(base: SolType, suffix: &str) -> Result<SolType, String> {
150        let suffix = suffix.trim();
151        if suffix.is_empty() {
152            return Ok(base);
153        }
154
155        if !suffix.starts_with('[') {
156            return Err(format!("Expected '[' but found: {}", suffix));
157        }
158
159        let close = suffix
160            .find(']')
161            .ok_or_else(|| format!("Missing ']' in: {}", suffix))?;
162        let size_str = &suffix[1..close];
163        let rest = &suffix[close + 1..];
164
165        let array_type = if size_str.is_empty() {
166            SolType::Array(Box::new(base))
167        } else {
168            let size: usize = size_str
169                .parse()
170                .map_err(|_| format!("Invalid array size: {}", size_str))?;
171            SolType::FixedArray(Box::new(base), size)
172        };
173
174        Self::apply_array_suffixes(array_type, rest)
175    }
176
177    fn find_matching_paren(s: &str, start: usize) -> Result<usize, String> {
178        let mut depth = 0;
179        for (i, ch) in s[start..].char_indices() {
180            match ch {
181                '(' => depth += 1,
182                ')' => {
183                    depth -= 1;
184                    if depth == 0 {
185                        return Ok(start + i);
186                    }
187                }
188                _ => {}
189            }
190        }
191        Err(format!("Unmatched parentheses in: {}", s))
192    }
193
194    pub fn canonical_signature(&self) -> String {
195        let params: Vec<String> = self.inputs.iter().map(|t| t.canonical_name()).collect();
196        format!("{}({})", self.name, params.join(","))
197    }

205    fn test_parse_simple() {
