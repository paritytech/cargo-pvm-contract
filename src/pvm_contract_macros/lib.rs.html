10/// Marks a module as a PVM smart contract, generating dispatch logic and entry points.
11///
12/// # Attributes
13///
14/// - `"path/to/Interface.sol"` - Optional Solidity interface file defining method signatures
15/// - `no_alloc` - Disables the allocator (uses fixed-size stack buffers)
16/// - `buffer = N` - Sets the calldata buffer size for no_alloc mode (default: 256)
17///
18/// # Usage with Solidity Interface (Recommended)
19///
20/// Create a Solidity interface file defining your contract's ABI:
21///
22/// ```solidity
23/// // MyToken.sol
24/// interface MyToken {
25///     function totalSupply() external view returns (uint256);
26///     function balanceOf(address account) external view returns (uint256);
27///     function transfer(address to, uint256 amount) external;
28/// }
29/// ```
30///
31/// Then implement the interface in Rust:
32///
33/// ```ignore
34/// use pvm_contract::{Address, U256};
35///
36/// #[pvm_contract::contract("MyToken.sol")]
37/// mod my_token {
38///     use super::*;
39///
40///     #[pvm_contract::constructor]
41///     pub fn new() -> Result<(), Error> { Ok(()) }
42///
43///     #[pvm_contract::method]
44///     pub fn total_supply() -> U256 { get_total_supply() }
45///
46///     #[pvm_contract::method]
47///     pub fn balance_of(account: Address) -> U256 { get_balance(&account) }
48///
49///     #[pvm_contract::method]
50///     pub fn transfer(to: Address, amount: U256) -> Result<(), Error> { Ok(()) }
51/// }
52/// ```
53///
54/// # Usage without Solidity Interface
55///
56/// You can also define contracts without a `.sol` file. Signatures are inferred from Rust types:
57///
58/// ```ignore
59/// use pvm_contract::{Address, U256};
60///
61/// #[pvm_contract::contract]
62/// mod my_token {
63///     use super::*;
64///
65///     #[pvm_contract::constructor]
66///     pub fn new() -> Result<(), Error> { Ok(()) }
67///
68///     #[pvm_contract::method]
69///     pub fn total_supply() -> U256 { U256::ZERO }
70///
71///     #[pvm_contract::method]
72///     pub fn balance_of(account: Address) -> U256 { U256::ZERO }
73///
74///     #[pvm_contract::method]
75///     pub fn transfer(to: Address, amount: U256) -> Result<(), Error> { Ok(()) }
76/// }
77/// ```
78///
79/// The builder will automatically generate an ABI JSON file alongside the `.polkavm` binary.
80///
81/// # Name Matching
82///
83/// Rust function names are converted to camelCase for Solidity compatibility:
84/// - `total_supply` → `totalSupply`
85/// - `balance_of` → `balanceOf`
86///
87/// For custom name mapping, use the `rename` attribute:
88///
89/// ```ignore
90/// #[pvm_contract::method(rename = "getBalance")]
91/// pub fn balance_of(account: Address) -> U256 { ... }
92/// ```
93///
94/// # Generated Code
95///
96/// ## Entry Points
97///
98/// The macro generates two PolkaVM entry points:
99///
100/// ```ignore
101/// #[no_mangle]
102/// #[polkavm_derive::polkavm_export]
103/// pub extern "C" fn deploy() { /* constructor logic */ }
104///
105/// #[no_mangle]
106/// #[polkavm_derive::polkavm_export]
107/// pub extern "C" fn call() { /* dispatch logic */ }
108/// ```
109///
110/// ## Error Type
111///
112/// An empty `Error` enum is generated inside the contract module. Add your own variants:
113///
114/// ```ignore
115/// mod my_token {
116///     #[derive(Debug, Clone, Copy, PartialEq, Eq)]
117///     pub enum Error {
118///         // Add your errors here:
119///         InsufficientBalance,
120///         Unauthorized,
121///     }
122///
123///     impl AsRef<[u8]> for Error {
124///         fn as_ref(&self) -> &[u8] {
125///             match self {
126///                 Self::InsufficientBalance => b"InsufficientBalance",
127///                 Self::Unauthorized => b"Unauthorized",
128///             }
129///         }
130///     }
131///     // ... methods
132/// }
133/// ```
134///
135/// ## Dispatch Logic (alloc mode)
136///
137/// With allocation enabled (default), the `call()` function uses `Vec`:
138///
139/// ```ignore
140/// pub extern "C" fn call() {
141///     let call_data_len = pvm_contract::api::call_data_size() as usize;
142///     let mut call_data = vec![0u8; call_data_len];
143///     pvm_contract::api::call_data_copy(&mut call_data, 0);
144///
145///     let result: Result<Option<Vec<u8>>, Vec<u8>> = (|| {
146///         if call_data.len() < 4 { return Err(b"UnknownSelector".to_vec()); }
147///         let selector: [u8; 4] = call_data[0..4].try_into().unwrap();
148///
149///         match selector {
150///             [0x18, 0x16, 0x0d, 0xdd] => {  // totalSupply()
151///                 Ok(Some(my_token::total_supply().to_be_bytes::<32>().to_vec()))
152///             }
153///             _ => Err(b"UnknownSelector".to_vec()),
154///         }
155///     })();
156///
157///     match result {
158///         Ok(Some(data)) => pvm_contract::api::return_value(pvm_contract::ReturnFlags::empty(), &data),
159///         Ok(None) => {}
160///         Err(data) => pvm_contract::api::return_value(pvm_contract::ReturnFlags::REVERT, &data),
161///     }
162/// }
163/// ```
164///
165/// ## Dispatch Logic (no_alloc mode)
166///
167/// With `no_alloc`, fixed-size stack buffers are used instead:
168///
169/// ```ignore
170/// #[pvm_contract::contract("MyToken.sol", no_alloc, buffer = 512)]
171/// mod my_token { /* ... */ }
172///
173/// // Expands to:
174/// pub extern "C" fn call() {
175///     let call_data_len = pvm_contract::api::call_data_size() as usize;
176///     let mut call_data = [0u8; 512];  // fixed buffer
177///
178///     if call_data_len > 512 {
179///         pvm_contract::api::return_value(pvm_contract::ReturnFlags::REVERT, b"CalldataTooLarge");
180///         return;
181///     }
182///     pvm_contract::api::call_data_copy(&mut call_data[..call_data_len], 0);
183///
184///     let result: Result<Option<&[u8]>, &[u8]> = (|| {
185///         // ... dispatch using slices instead of Vec
186///     })();
187///     // ...
188/// }
189/// ```
190///
191/// # Return Type Flexibility
192///
193/// Methods can return either:
194/// - `Result<T, Error>` - For fallible operations that may revert
195/// - `T` - For infallible operations (macro wraps in `Ok(...)`)
196#[proc_macro_attribute]
197pub fn contract(attr: TokenStream, item: TokenStream) -> TokenStream {
198    let args = parse_macro_input!(attr as codegen::ContractArgs);
199    let input = parse_macro_input!(item as ItemMod);
200
201    match codegen::expand_contract(args, input) {
202        Ok(tokens) => tokens.into(),
203        Err(err) => err.to_compile_error().into(),
204    }
205}

207/// Marks a function as a contract method. The signature is derived from the Solidity interface file.
208///
209/// # Attributes
210///
211/// - `rename = "name"` - Override the Solidity function name to match (default: snake_case conversion)
212///
213/// # Name Matching
214///
215/// By default, the macro converts the Rust function name from snake_case to camelCase
216/// to match the Solidity function:
217///
218/// ```ignore
219/// #[pvm_contract::method]
220/// pub fn total_supply() -> U256 { ... }  // matches totalSupply()
221/// ```
222///
223/// Use `rename` when the naming convention differs:
224///
225/// ```ignore
226/// #[pvm_contract::method(rename = "getBalance")]
227/// pub fn balance_of(account: Address) -> U256 { ... }  // matches getBalance(address)
228/// ```
229///
230/// # Return Types
231///
232/// Methods support two return patterns:
233///
234/// ```ignore
235/// // Fallible - can revert with error
236/// #[pvm_contract::method]
237/// pub fn transfer(to: Address, amount: U256) -> Result<(), Error> { ... }
238///
239/// // Infallible - always succeeds
240/// #[pvm_contract::method]
241/// pub fn balance_of(account: Address) -> U256 { ... }
242/// ```
243#[proc_macro_attribute]
244pub fn method(attr: TokenStream, item: TokenStream) -> TokenStream {
245    let args = parse_macro_input!(attr as codegen::MethodArgs);
246    let input = parse_macro_input!(item as ItemFn);
247
248    match codegen::expand_method(args, input) {
249        Ok(tokens) => tokens.into(),
250        Err(err) => err.to_compile_error().into(),
251    }
252}

254/// Marks a function as the contract constructor, called during deployment.
255///
256/// # Example
257///
258/// ```ignore
259/// #[pvm_contract::constructor]
260/// pub fn new() -> Result<(), Error> {
261///     set_owner(pvm_contract::caller());
262///     Ok(())
263/// }
264/// ```
265///
266/// Must return `Result<(), Error>`. Returning `Err` reverts the deployment.
267#[proc_macro_attribute]
268pub fn constructor(_attr: TokenStream, item: TokenStream) -> TokenStream {
269    let input = parse_macro_input!(item as ItemFn);
270
271    match codegen::expand_constructor(input) {
272        Ok(tokens) => tokens.into(),
273        Err(err) => err.to_compile_error().into(),
274    }
275}

277/// Marks a function as the fallback handler for unknown selectors.
278///
279/// Called when:
280/// - Calldata is less than 4 bytes
281/// - No method matches the selector
282///
283/// # Example
284///
285/// ```ignore
286/// #[pvm_contract::fallback]
287/// pub fn fallback() -> Result<(), Error> {
288///     Err(Error::UnknownSelector)
289/// }
290/// ```
291///
292/// Must return `Result<(), Error>`. Commonly used to reject unknown calls.
293#[proc_macro_attribute]
294pub fn fallback(_attr: TokenStream, item: TokenStream) -> TokenStream {
295    let input = parse_macro_input!(item as ItemFn);
296
297    match codegen::expand_fallback(input) {
298        Ok(tokens) => tokens.into(),
299        Err(err) => err.to_compile_error().into(),
300    }
