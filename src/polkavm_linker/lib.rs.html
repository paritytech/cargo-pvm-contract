24struct VersionDetector {
25    major: u32,
26    minor: u32,
27    date: u32,
28    is_nightly: bool,
29}
30
31const YEAR: u32 = 10000;
32const MONTH: u32 = 100;
33
34impl VersionDetector {
35    fn new() -> Result<Self, String> {
36        let rustc = std::env::var("RUSTC").unwrap_or_else(|_| "rustc".to_owned());
37        let version = std::process::Command::new(&rustc)
38            .arg("--version")
39            .output()
40            .map_err(|err| format!("failed to detect rustc version: failed to run '{rustc} --version': {err}"))?
41            .stdout;
42
43        let full_version = String::from_utf8(version)
44            .map_err(|_| format!("failed to detect rustc version: '{rustc} --version' returned non-valid UTF-8"))?;
45        let full_version = full_version.trim();
46        // Examples of version strings:
47        //   'rustc 1.86.0 (05f9846f8 2025-03-31)'
48        //   'rustc 1.90.0-nightly (e9182f195 2025-07-13)'
49
50        (move || -> Option<Self> {
51            let version = full_version.split(' ').nth(1)?;
52            let is_nightly = full_version.contains("-nightly");
53            let date = {
54                let mut it = full_version[full_version.rfind(' ')? + 1..full_version.len() - 1].split('-');
55                let year: u32 = it.next()?.parse().ok()?;
56                let month: u32 = it.next()?.parse().ok()?;
57                let day: u32 = it.next()?.parse().ok()?;
58                year * YEAR + month * MONTH + day
59            };
60            let mut it = version.split('.');
61            let major: u32 = it.next()?.parse().ok()?;
62            let minor: u32 = it.next()?.parse().ok()?;
63
64            Some(Self {
65                major,
66                minor,
67                date,
68                is_nightly,
69            })
70        })()
71        .ok_or_else(|| format!("failed to detect rustc version: failed to parse output of '{rustc}' --version: {full_version:?}"))
72    }
73
74    fn check_feature(&self, req_minor: u32, year: u32, month: u32, day: u32) -> bool {
75        self.major > 1
76            || (self.major == 1 && self.minor >= req_minor && !self.is_nightly)
77            || (self.major == 1 && self.minor >= (req_minor + 1))
78            || (self.major == 1 && self.is_nightly && self.date >= year * YEAR + month * MONTH + day)
79    }
80}

82fn target_json_path_impl(
83    json_name: &str,
84    json_contents_old: &str,
85    json_contents_new: &str,
86    rustc_version: RustcVersion,
87) -> Result<PathBuf, String> {
88    let rustc_version = match rustc_version {
89        RustcVersion::Autodetect => VersionDetector::new()?,
90        RustcVersion::Rustc_1_91 => VersionDetector {
91            major: 1,
92            minor: 91,
93            date: 2025 * YEAR + 10 * MONTH + 30,
94            is_nightly: false,
95        },
96        RustcVersion::Legacy => VersionDetector {
97            major: 1,
98            minor: 90,
99            date: 2025 * YEAR + 9 * MONTH + 18,
100            is_nightly: false,
101        },
102    };
103
104    // https://github.com/rust-lang/rust/pull/144443
105    let (json_contents, subdirectory) = if rustc_version.check_feature(91, 2025, 9, 1) {
106        (json_contents_new, "1_91")
107    } else {
108        (json_contents_old, "legacy")
109    };
110
111    let version = env!("CARGO_PKG_VERSION");
112    let cache_path = dirs::cache_dir()
113        .or_else(|| std::env::current_dir().ok())
114        .unwrap_or_else(|| PathBuf::from("./"))
115        .join(".polkavm-linker")
116        .join(version)
117        .join(subdirectory);
118
119    if let Err(error) = std::fs::create_dir_all(&cache_path) {
120        return Err(format!(
121            "failed to fetch path to the PolkaVM target JSON: failed to create {cache_path:?}: {error}"
122        ));
123    }
124
125    let json_path = cache_path.join(json_name);
126
127    // If the file exists only write if the contents changed, to not unnecesarily bump the file's mtime.
128    if std::fs::read_to_string(&json_path)
129        .ok()
130        .map_or(true, |existing_contents| existing_contents != json_contents)
131    {
132        std::fs::write(&json_path, json_contents)
133            .map_err(|error| format!("failed to fetch path to the PolkaVM target JSON: failed to write to {json_path:?}: {error}"))?;
134    }
135
136    json_path
137        .canonicalize()
138        .map_err(|error| format!("failed to fetch path to the PolkaVM target JSON: failed to canonicalize path {json_path:?}: {error}"))
139}

141#[allow(non_camel_case_types)]

143#[non_exhaustive]
144pub enum RustcVersion {
145    #[default]
146    Autodetect,
147    Legacy,

151#[derive(Clone)]
152#[non_exhaustive]
153pub struct TargetJsonArgs {
154    pub rustc_version: RustcVersion,
155    pub is_64_bit: bool,
156}
157
158impl Default for TargetJsonArgs {
159    fn default() -> Self {
160        TargetJsonArgs {
161            rustc_version: Default::default(),
162            is_64_bit: true,
163        }
164    }
165}
166
167pub fn target_json_path(args: TargetJsonArgs) -> Result<PathBuf, String> {
168    if args.is_64_bit {
169        target_json_path_impl(
170            "riscv64emac-unknown-none-polkavm.json",
171            TARGET_JSON_64_BIT_OLD,
172            TARGET_JSON_64_BIT_NEW,
173            args.rustc_version,
174        )
175    } else {
176        target_json_path_impl(
177            "riscv32emac-unknown-none-polkavm.json",
178            TARGET_JSON_32_BIT_OLD,
179            TARGET_JSON_32_BIT_NEW,
180            args.rustc_version,
181        )
182    }
183}
