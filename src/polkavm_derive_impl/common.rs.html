15pub fn is_path_eq(path: &syn::Path, ident: &str) -> bool {
16    let segments: Vec<_> = ident.split("::").collect();
17    path.segments.len() == segments.len()
18        && path
19            .segments
20            .iter()
21            .zip(segments.iter())
22            .all(|(segment, expected)| segment.ident == expected && segment.arguments.is_none())
23}

25pub fn is_doc(attr: &syn::Attribute) -> bool {

33pub fn is_cfg(attr: &syn::Attribute) -> bool {

35        is_path_eq(path, "cfg") || is_path_eq(path, "cfg_attr") || is_path_eq(path, "allow") || is_path_eq(path, "deny")
36    } else {
37        false
38    }
