20// Tokens to identify file descriptor
21const TTY_TOKEN: Token = Token(0);
22const SIGNAL_TOKEN: Token = Token(1);
23#[cfg(feature = "event-stream")]
24const WAKE_TOKEN: Token = Token(2);

26// I (@zrzka) wasn't able to read more than 1_022 bytes when testing
27// reading on macOS/Linux -> we don't need bigger buffer and 1k of bytes
28// is enough.
29const TTY_BUFFER_SIZE: usize = 1_024;
30
31pub(crate) struct UnixInternalEventSource {
32    poll: Poll,
33    events: Events,
34    parser: Parser,
35    tty_buffer: [u8; TTY_BUFFER_SIZE],

42impl UnixInternalEventSource {
43    pub fn new() -> Result<Self> {
44        UnixInternalEventSource::from_file_descriptor(tty_fd()?)
45    }
46
47    pub(crate) fn from_file_descriptor(input_fd: FileDesc) -> Result<Self> {
48        let poll = Poll::new()?;
49        let registry = poll.registry();
50
51        let tty_raw_fd = input_fd.raw_fd();
52        let mut tty_ev = SourceFd(&tty_raw_fd);
53        registry.register(&mut tty_ev, TTY_TOKEN, Interest::READABLE)?;
54
55        let mut signals = Signals::new(&[signal_hook::consts::SIGWINCH])?;
56        registry.register(&mut signals, SIGNAL_TOKEN, Interest::READABLE)?;
57
58        #[cfg(feature = "event-stream")]
59        let waker = Waker::new(registry, WAKE_TOKEN)?;
60
61        Ok(UnixInternalEventSource {
62            poll,
63            events: Events::with_capacity(3),
64            parser: Parser::default(),
65            tty_buffer: [0u8; TTY_BUFFER_SIZE],
66            tty_fd: input_fd,
67            signals,
68            #[cfg(feature = "event-stream")]
69            waker,
70        })
71    }
72}

74impl EventSource for UnixInternalEventSource {
75    fn try_read(&mut self, timeout: Option<Duration>) -> Result<Option<InternalEvent>> {
76        if let Some(event) = self.parser.next() {
77            return Ok(Some(event));
78        }
79
80        let timeout = PollTimeout::new(timeout);
81
82        loop {
83            if let Err(e) = self.poll.poll(&mut self.events, timeout.leftover()) {
84                // Mio will throw an interrupted error in case of cursor position retrieval. We need to retry until it succeeds.
85                // Previous versions of Mio (< 0.7) would automatically retry the poll call if it was interrupted (if EINTR was returned).
86                // https://docs.rs/mio/0.7.0/mio/struct.Poll.html#notes
87                if e.kind() == io::ErrorKind::Interrupted {
88                    continue;
89                } else {
90                    return Err(e);
91                }
92            };
93
94            if self.events.is_empty() {
95                // No readiness events = timeout
96                return Ok(None);
97            }
98
99            for token in self.events.iter().map(|x| x.token()) {
100                match token {
101                    TTY_TOKEN => {
102                        loop {
103                            match self.tty_fd.read(&mut self.tty_buffer, TTY_BUFFER_SIZE) {
104                                Ok(read_count) => {
105                                    if read_count > 0 {
106                                        self.parser.advance(
107                                            &self.tty_buffer[..read_count],
108                                            read_count == TTY_BUFFER_SIZE,
109                                        );
110                                    }
111                                }
112                                Err(e) => {
113                                    // No more data to read at the moment. We will receive another event
114                                    if e.kind() == io::ErrorKind::WouldBlock {
115                                        break;
116                                    }
117                                    // once more data is available to read.
118                                    else if e.kind() == io::ErrorKind::Interrupted {
119                                        continue;
120                                    }
121                                }
122                            };
123
124                            if let Some(event) = self.parser.next() {
125                                return Ok(Some(event));
126                            }
127                        }
128                    }
129                    SIGNAL_TOKEN => {
130                        for signal in self.signals.pending() {
131                            match signal {
132                                signal_hook::consts::SIGWINCH => {
133                                    // TODO Should we remove tput?
134                                    //
135                                    // This can take a really long time, because terminal::size can
136                                    // launch new process (tput) and then it parses its output. It's
137                                    // not a really long time from the absolute time point of view, but
138                                    // it's a really long time from the mio, async-std/tokio executor, ...
139                                    // point of view.
140                                    let new_size = crate::terminal::size()?;
141                                    return Ok(Some(InternalEvent::Event(Event::Resize(
142                                        new_size.0, new_size.1,
143                                    ))));
144                                }
145                                _ => unreachable!("Synchronize signal registration & handling"),
146                            };
147                        }
148                    }
149                    #[cfg(feature = "event-stream")]
150                    WAKE_TOKEN => {
151                        return Err(std::io::Error::new(
152                            std::io::ErrorKind::Interrupted,
153                            "Poll operation was woken up by `Waker::wake`",
154                        ));
155                    }
156                    _ => unreachable!("Synchronize Evented handle registration & token handling"),
157                }
158            }
159
160            // Processing above can take some time, check if timeout expired
161            if timeout.elapsed() {
162                return Ok(None);
163            }
164        }
165    }
166
167    #[cfg(feature = "event-stream")]
168    fn waker(&self) -> Waker {
169        self.waker.clone()
170    }
171}

173//
174// Following `Parser` structure exists for two reasons:
175//
176//  * mimic anes Parser interface
177//  * move the advancing, parsing, ... stuff out of the `try_read` method
178//
179#[derive(Debug)]
180struct Parser {
181    buffer: Vec<u8>,
182    internal_events: VecDeque<InternalEvent>,
183}
184
185impl Default for Parser {
186    fn default() -> Self {
187        Parser {
188            // This buffer is used for -> 1 <- ANSI escape sequence. Are we
189            // aware of any ANSI escape sequence that is bigger? Can we make
190            // it smaller?
191            //
192            // Probably not worth spending more time on this as "there's a plan"
193            // to use the anes crate parser.
194            buffer: Vec::with_capacity(256),
195            // TTY_BUFFER_SIZE is 1_024 bytes. How many ANSI escape sequences can
196            // fit? What is an average sequence length? Let's guess here
197            // and say that the average ANSI escape sequence length is 8 bytes. Thus
198            // the buffer size should be 1024/8=128 to avoid additional allocations
199            // when processing large amounts of data.
200            //
201            // There's no need to make it bigger, because when you look at the `try_read`
202            // method implementation, all events are consumed before the next TTY_BUFFER
203            // is processed -> events pushed.
204            internal_events: VecDeque::with_capacity(128),
205        }
206    }
207}
208
209impl Parser {
210    fn advance(&mut self, buffer: &[u8], more: bool) {
211        for (idx, byte) in buffer.iter().enumerate() {
212            let more = idx + 1 < buffer.len() || more;
213
214            self.buffer.push(*byte);
215
216            match parse_event(&self.buffer, more) {
217                Ok(Some(ie)) => {
218                    self.internal_events.push_back(ie);
219                    self.buffer.clear();
220                }
221                Ok(None) => {
222                    // Event can't be parsed, because we don't have enough bytes for
223                    // the current sequence. Keep the buffer and process next bytes.
224                }
225                Err(_) => {
226                    // Event can't be parsed (not enough parameters, parameter is not a number, ...).
227                    // Clear the buffer and continue with another sequence.
228                    self.buffer.clear();
229                }
230            }
231        }
232    }
233}
234
235impl Iterator for Parser {
236    type Item = InternalEvent;
237
238    fn next(&mut self) -> Option<Self::Item> {
239        self.internal_events.pop_front()
240    }
