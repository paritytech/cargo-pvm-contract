16// Some(Termios) -> we're in the raw mode and this is the previous mode
17// None -> we're not in the raw mode
18static TERMINAL_MODE_PRIOR_RAW_MODE: Mutex<Option<Termios>> = parking_lot::const_mutex(None);
19
20pub(crate) fn is_raw_mode_enabled() -> bool {
21    TERMINAL_MODE_PRIOR_RAW_MODE.lock().is_some()
22}
23
24#[allow(clippy::useless_conversion)]
25pub(crate) fn size() -> Result<(u16, u16)> {
26    // http://rosettacode.org/wiki/Terminal_control/Dimensions#Library:_BSD_libc
27    let mut size = winsize {
28        ws_row: 0,
29        ws_col: 0,
30        ws_xpixel: 0,
31        ws_ypixel: 0,
32    };
33
34    let file = File::open("/dev/tty").map(|file| (FileDesc::new(file.into_raw_fd(), true)));
35    let fd = if let Ok(file) = &file {
36        file.raw_fd()
37    } else {
38        // Fallback to libc::STDOUT_FILENO if /dev/tty is missing
39        STDOUT_FILENO
40    };
41
42    if wrap_with_result(unsafe { ioctl(fd, TIOCGWINSZ.into(), &mut size) }).is_ok()
43        && size.ws_col != 0
44        && size.ws_row != 0
45    {
46        return Ok((size.ws_col, size.ws_row));
47    }
48
49    tput_size().ok_or_else(|| std::io::Error::last_os_error().into())
50}

52pub(crate) fn enable_raw_mode() -> Result<()> {
53    let mut original_mode = TERMINAL_MODE_PRIOR_RAW_MODE.lock();
54
55    if original_mode.is_some() {
56        return Ok(());
57    }
58
59    let tty = tty_fd()?;
60    let fd = tty.raw_fd();
61    let mut ios = get_terminal_attr(fd)?;
62    let original_mode_ios = ios;
63
64    raw_terminal_attr(&mut ios);
65    set_terminal_attr(fd, &ios)?;
66
67    // Keep it last - set the original mode only if we were able to switch to the raw mode
68    *original_mode = Some(original_mode_ios);
69
70    Ok(())
71}

73/// Reset the raw mode.
74///
75/// More precisely, reset the whole termios mode to what it was before the first call
76/// to [enable_raw_mode]. If you don't mess with termios outside of crossterm, it's
77/// effectively disabling the raw mode and doing nothing else.
78pub(crate) fn disable_raw_mode() -> Result<()> {
79    let mut original_mode = TERMINAL_MODE_PRIOR_RAW_MODE.lock();
80
81    if let Some(original_mode_ios) = original_mode.as_ref() {
82        let tty = tty_fd()?;
83        set_terminal_attr(tty.raw_fd(), original_mode_ios)?;
84        // Keep it last - remove the original mode only if we were able to switch back
85        *original_mode = None;
86    }
87
88    Ok(())
89}

91/// execute tput with the given argument and parse
92/// the output as a u16.
93///
94/// The arg should be "cols" or "lines"
95fn tput_value(arg: &str) -> Option<u16> {
96    let output = process::Command::new("tput").arg(arg).output().ok()?;
97    let value = output
98        .stdout
99        .into_iter()
100        .filter_map(|b| char::from(b).to_digit(10))
101        .fold(0, |v, n| v * 10 + n as u16);
102
103    if value > 0 {
104        Some(value)
105    } else {
106        None
107    }
108}

110/// Returns the size of the screen as determined by tput.
111///
112/// This alternate way of computing the size is useful
113/// when in a subshell.
114fn tput_size() -> Option<(u16, u16)> {
115    match (tput_value("cols"), tput_value("lines")) {
116        (Some(w), Some(h)) => Some((w, h)),
117        _ => None,
118    }
119}

121// Transform the given mode into an raw mode (non-canonical) mode.
122fn raw_terminal_attr(termios: &mut Termios) {
123    unsafe { cfmakeraw(termios) }
124}
125
126fn get_terminal_attr(fd: RawFd) -> Result<Termios> {
127    unsafe {
128        let mut termios = mem::zeroed();
129        wrap_with_result(tcgetattr(fd, &mut termios))?;
130        Ok(termios)
131    }
132}
133
134fn set_terminal_attr(fd: RawFd, termios: &Termios) -> Result<()> {
135    wrap_with_result(unsafe { tcsetattr(fd, TCSANOW, termios) })
136}
137
138fn wrap_with_result(result: i32) -> Result<()> {
139    if result == -1 {
140        Err(io::Error::last_os_error())
141    } else {
142        Ok(())
143    }
