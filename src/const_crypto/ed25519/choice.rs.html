1/// The `Choice` struct represents a choice for use in conditional
2/// assignment.
3///
4/// It is a wrapper around a `u8`, which should have the value either
5/// `1` (true) or `0` (false).
6///
7/// The conversion from `u8` to `Choice` passes the value through an
8/// optimization barrier, as a best-effort attempt to prevent the
9/// compiler from inferring that the `Choice` value is a boolean. This
10/// strategy is based on Tim Maclean's
11/// [work on `rust-timing-shield`][rust-timing-shield], which attempts
12/// to provide a more comprehensive approach for preventing software
13/// side-channels in Rust code.
14///
15/// The `Choice` struct implements operators for AND, OR, XOR, and NOT,
16/// to allow combining `Choice` values. These operations do not
17/// short-circuit.
18///
19/// [rust-timing-shield]:
20/// https://www.chosenplaintext.ca/open-source/rust-timing-shield/security
21#[derive(Copy, Clone, Debug)]
22pub struct Choice(u8);
23
24impl Choice {
25    /// Unwrap the `Choice` wrapper to reveal the underlying `u8`.
26    ///
27    /// # Note
28    ///
29    /// This function only exists as an **escape hatch** for the rare
30    /// case where it's not possible to use one of the
31    /// `subtle`-provided trait impls.
32    ///
33    /// **To convert a `Choice` to a `bool`, use the `From`
34    /// implementation instead.**
35    #[inline]
36    pub const fn unwrap_u8(&self) -> u8 {
37        self.0
38    }
39
40    pub const fn into(self) -> bool {
41        self.0 != 0
42    }
43
44    #[inline]
45    pub(crate) const fn from_u8(input: u8) -> Choice {
46        Choice(input)
47    }
48}
49
50impl From<Choice> for bool {
51    /// Convert the `Choice` wrapper into a `bool`, depending on whether
52    /// the underlying `u8` was a `0` or a `1`.
53    ///
54    /// # Note
55    ///
56    /// This function exists to avoid having higher-level cryptographic
57    /// protocol implementations duplicating this pattern.
58    ///
59    /// The intended use case for this conversion is at the _end_ of a
60    /// higher-level primitive implementation: for example, in checking
61    /// a keyed MAC, where the verification should happen in
62    /// constant-time (and thus use a `Choice`) but it is safe to
63    /// return a `bool` at the end of the verification.
64    #[inline]
65    fn from(source: Choice) -> bool {
66        debug_assert!((source.0 == 0u8) | (source.0 == 1u8));
67        source.0 != 0
68    }
69}
