10/// Default encoding function.
11///
12/// # Safety
13///
14/// `output` must be at least `2 * input.len()` bytes long.
15pub(crate) unsafe fn encode<const UPPER: bool>(input: &[u8], mut output: impl Output) {
16    for &byte in input {
17        let (high, low) = byte2hex::<UPPER>(byte);
18        output.write_byte(high);
19        output.write_byte(low);
20    }
21}

23/// Encodes unaligned chunks of `T` in `input` to `output` using `encode_chunk`.
24///
25/// The remainder is encoded using the generic [`encode`].
26#[inline]
27#[allow(dead_code)]
28pub(crate) unsafe fn encode_unaligned_chunks<const UPPER: bool, T: Copy, U: Copy>(
29    input: &[u8],
30    mut output: impl Output,
31    mut encode_chunk: impl FnMut(T) -> U,
32) {
33    debug_assert_eq!(size_of::<U>(), size_of::<T>() * 2);
34    let (chunks, remainder) = chunks_unaligned::<T>(input);
35    for chunk in chunks {
36        output.write(as_bytes(&encode_chunk(chunk)));
37    }
38    unsafe { encode::<UPPER>(remainder, output) };
39}

41/// Default check function.
42#[inline]
43pub(crate) const fn check(mut input: &[u8]) -> bool {
44    while let &[byte, ref rest @ ..] = input {
45        if HEX_DECODE_LUT[byte as usize] == NIL {
46            return false;
47        }
48        input = rest;
49    }
50    true
51}

53/// Runs the given check function on unaligned chunks of `T` in `input`, with the remainder passed
54/// to the generic [`check`].
55#[inline]
56#[allow(dead_code)]
57pub(crate) fn check_unaligned_chunks<T: Copy>(
58    input: &[u8],
59    check_chunk: impl FnMut(T) -> bool,
60) -> bool {
61    let (mut chunks, remainder) = chunks_unaligned(input);
62    chunks.all(check_chunk) && check(remainder)
63}

65/// Default checked decoding function.
66///
67/// # Safety
68///
69/// Assumes `output.len() == input.len() / 2`.
70pub(crate) unsafe fn decode_checked(input: &[u8], output: &mut [u8]) -> bool {
71    unsafe { decode_maybe_check::<true>(input, output) }
72}

74/// Default unchecked decoding function.
75///
76/// # Safety
77///
78/// Assumes `output.len() == input.len() / 2` and that the input is valid hex.
79pub(crate) unsafe fn decode_unchecked(input: &[u8], output: impl Output) {
80    #[allow(unused_braces)] // False positive on older rust versions.
81    let success = unsafe { decode_maybe_check::<{ cfg!(debug_assertions) }>(input, output) };
82    debug_assert!(success);
83}

85/// Default decoding function. Checks input validity if `CHECK` is `true`, otherwise assumes it.
86///
87/// # Safety
88///
89/// Assumes `output.len() == input.len() / 2` and that the input is valid hex if `CHECK` is `true`.
90#[inline(always)]
91unsafe fn decode_maybe_check<const CHECK: bool>(input: &[u8], mut output: impl Output) -> bool {
92    macro_rules! next {
93        ($var:ident, $i:expr) => {
94            let hex = unsafe { *input.get_unchecked($i) };
95            let $var = HEX_DECODE_LUT[hex as usize];
96            if CHECK {
97                if $var == NIL {
98                    return false;
99                }
100            }
101        };
102    }
103
104    let l = output.remaining().unwrap_or(input.len() / 2);
105    debug_assert_eq!(l, input.len() / 2);
106    let mut i = 0;
107    while i < l {
108        next!(high, i * 2);
109        next!(low, i * 2 + 1);
110        output.write_byte(high << 4 | low);
111        i += 1;
112    }
113    true
114}

116/// Decodes unaligned chunks of `U` in `input` to `output` using `decode_chunk`.
117///
118/// The remainder is decoded using the generic [`decode_unchecked`].
119#[inline]
120#[allow(dead_code)]
121pub(crate) unsafe fn decode_unchecked_unaligned_chunks<T: Copy, U: Copy>(
122    input: &[u8],
123    mut output: impl Output,
124    mut decode_chunk: impl FnMut(U) -> T,
125) {
126    debug_assert_eq!(size_of::<U>(), size_of::<T>() * 2);
127    let (chunks, remainder) = chunks_unaligned::<U>(input);
128    for chunk in chunks {
129        output.write(as_bytes(&decode_chunk(chunk)));
130    }
131    unsafe { decode_unchecked(remainder, output) };
132}

134#[inline]
135fn chunks_unaligned<T: Copy>(input: &[u8]) -> (impl ExactSizeIterator<Item = T> + '_, &[u8]) {
136    let chunks = input.chunks_exact(core::mem::size_of::<T>());
137    let remainder = chunks.remainder();
138    (
139        chunks.map(|chunk| unsafe { chunk.as_ptr().cast::<T>().read_unaligned() }),
140        remainder,
141    )
142}
